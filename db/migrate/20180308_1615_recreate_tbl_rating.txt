CREATE TABLE nh_rating
(
  rating_id           INT AUTO_INCREMENT
    PRIMARY KEY,
  rating_name         VARCHAR(255)           NULL,
  rating_label        VARCHAR(255)           NULL
  COMMENT 'text hoặc biểu tượng hiển thị lúc đánh giá',
  rating_value        FLOAT DEFAULT '100'    NOT NULL
  COMMENT 'Phân trăm huê hồng được hưởng',
  rating_status       TINYINT(1) DEFAULT '0' NOT NULL
  COMMENT '0: hide, 1: show',
  rating_deleted      TINYINT(1) DEFAULT '0' NOT NULL,
  rating_order        INT DEFAULT '0'        NULL
  COMMENT 'Thứ tự sắp xếp',
  user_id             INT DEFAULT '0'        NOT NULL,
  rating_created_time INT DEFAULT '0'        NOT NULL,
  rating_updated_time INT DEFAULT '0'        NULL,
  rating_token_key    CHAR(32)               NULL
)
  ENGINE = InnoDB;
