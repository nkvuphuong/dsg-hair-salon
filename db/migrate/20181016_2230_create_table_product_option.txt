CREATE TABLE nh_product_option
(
    product_option_id int(11) PRIMARY KEY AUTO_INCREMENT,
    product_group_id int(11) DEFAULT 0 COMMENT 'Nhóm sản phẩm',
    option_group_id int(11) DEFAULT 0 COMMENT 'Nhóm tuỳ chọn',
    product_option_time int(11) DEFAULT 0,
    product_option_time_update int(11) DEFAULT 0,
    product_option_deleted tinyint(1) DEFAULT 0
);
CREATE UNIQUE INDEX nh_product_option_product_option_id_uindex ON nh_product_option (product_option_id);
